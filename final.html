Model A equations

modelA <- function(time, state, parameters) {

    R <- state[1]
    P <- state[2]
    T <- state[3]

    # Resident whales (competition with P)
    dR <- r_R * R * (1 - (R + alpha * P)/K_R)

    # Pinnipeds (competition + predation)
    dP <- r_P * P * (1 - (P + beta * R)/K_P) - C * P * T

    # Transient whales (predator)
    dT <- B * C * P * T - D_T * T

    return(list(c(dR, dP, dT)))
}

library(deSolve)
require(deSolve)

# predator-prey model
LV <- function(time, state, para) {
    B_R = para[1]
    C_r = para[2]
    D_R = para[3]
    B_T = para[4]
    C_T = para[5]
    D_T = para[6]
    r_P = para[7]
    K_P = para[8]
    a = para[9]
    r_S = para[10]
    K_S = para[11]
    b = para[12]
    # state variables
    R = state[1]
    T = state[2]
    P = state[3]
    S = state[4]
    # equation
    X1 = R * (B_R * C_R * S - D_R)
    X2 = T * (B_T * C_T * P - D_T)
    X3 = r_P * P * (1 - P/K_P - a * S/K_P) - C_T * P * T
    X4 = r_S * S * (1 - S/K_S - b * P/K_S) - C_R * S * R

    return(list(c(X1, X2, X3, X4)))
}

    b)
